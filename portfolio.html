The idea

A model proposes. Something else decides.

This is the principle behind NEXUS CORE, written as a few lines in two languages.

authorize.py
from dataclasses import dataclass

@dataclass
class Action:
    name: str
    sensitive: bool = False

# A model may propose an action. It may not authorize itself.
def authorize(action, approved=False):
    return not action.sensitive or approved

plan = [Action("read_file"), Action("send_payment", sensitive=True)]

for step in plan:
    verdict = "run" if authorize(step) else "needs approval"
    print(step.name, "->", verdict)
#include <iostream>
#include <string>
#include <vector>

struct Action {
    std::string name;
    bool sensitive = false;
};

// A model may propose an action. It may not authorize itself.
bool authorize(const Action& a, bool approved = false) {
    return !a.sensitive || approved;
}

int main() {
    std::vector<Action> plan = {{"read_file"}, {"send_payment", true}};

    for (const auto& step : plan) {
        const char* verdict = authorize(step) ? "run" : "needs approval";
        std::cout << step.name << " -> " << verdict << "\n";
    }
}
